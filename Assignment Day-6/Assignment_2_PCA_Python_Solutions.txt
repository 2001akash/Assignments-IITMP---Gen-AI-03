QUESTION 1: Calculate Mean and Center Data
-------------------------------------------
Solution:

import numpy as np

    [2, 1],
    [3, 2],
    [4, 3],
    [5, 4],
    [6, 5]

QUESTION 3: Calculate Covariance Matrix
----------------------------------------
Solution:

import numpy as np

# Using centered data from Question 1
X = np.array([
    [2, 1],
    [3, 2],
    [4, 3],
    [5, 4],
    [6, 5]
])

mean = np.mean(X, axis=0)
X_centered = X - mean

# Calculate covariance matrix
cov_matrix = np.cov(X_centered.T)
print("Covariance matrix:")
print(cov_matrix)
# Output:
# [[2.5  2.5]
#  [2.5  2.5]]

# Extract variance of each feature (diagonal elements)
variance_feature1 = cov_matrix[0, 0]
variance_feature2 = cov_matrix[1, 1]
print("\nVariance of Feature 1:", variance_feature1)
# Output: 2.5
print("Variance of Feature 2:", variance_feature2)
# Output: 2.5

# Covariance between features (off-diagonal)
covariance = cov_matrix[0, 1]
print("Covariance between features:", covariance)
# Output: 2.5 (strong positive correlation)

QUESTION 9: Sort Eigenvalues and Eigenvectors
----------------------------------------------
Solution:

import numpy as np

C = np.array([
    [6, 4],
    [4, 3]
])

# Step 1: Find eigenvalues and eigenvectors
eigenvalues, eigenvectors = np.linalg.eig(C)

print("Original eigenvalues:", eigenvalues)
print("Original eigenvectors:")
print(eigenvectors)

# Step 2: Sort from largest to smallest
sorted_indices = np.argsort(eigenvalues)[::-1]
sorted_eigenvalues = eigenvalues[sorted_indices]
sorted_eigenvectors = eigenvectors[:, sorted_indices]

print(sorted_eigenvectors)

# Step 3: Interpretation

QUESTION 12: Parameterized PCA Function
----------------------------------------
Solution:

import numpy as np

def custom_pca(X, n_components):
    """
    Custom PCA implementation
    """
    # Step 1: Mean centering
    mean = np.mean(X, axis=0)
    X_centered = X - mean
    
    # Step 2: Covariance matrix
    cov_matrix = np.cov(X_centered.T)
    
    # Step 3: Eigenvalues and eigenvectors
    eigenvalues, eigenvectors = np.linalg.eig(cov_matrix)
    
    # Step 4: Sort by eigenvalues
    idx = np.argsort(eigenvalues)[::-1]
    eigenvalues = eigenvalues[idx]
    eigenvectors = eigenvectors[:, idx]
    
    # Step 5: Select n_components eigenvectors
    principal_components = eigenvectors[:, :n_components]
    
    # Step 6: Transform data
    X_pca = np.dot(X_centered, principal_components)
    
    return X_pca, principal_components, eigenvalues[:n_components]

# Test
X = np.array([
    [2, 1],
    [3, 2],
    [4, 3],
    [5, 4],
    [6, 5]
])

X_reduced = custom_pca(X, n_components=1)
print("Reduced data:")
print(X_reduced[0])

QUESTION 15: Explained Variance Analysis
-----------------------------------------
Solution:

import numpy as np
from sklearn.decomposition import PCA

X = np.array([
    [2, 1, 5],
    [3, 2, 6],
    [4, 3, 7],
    [5, 4, 8],
    [6, 5, 9]
])

# Apply PCA with all components
pca = PCA()
pca.fit(X)

# Explained variance
explained_var = pca.explained_variance_ratio_
cumulative_var = np.cumsum(explained_var)

print("Explained variance ratio:")
print(explained_var)

print("\nCumulative explained variance:")
print(cumulative_var)

print(f"\n1 component retains: {cumulative_var[0]:.2%}")
print(f"2 components retain: {cumulative_var[1]:.2%}")

QUESTION 18: Standardization Comparison
----------------------------------------
Solution:

import numpy as np
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt

X = np.array([
    [1, 100],
    [2, 200],
    [3, 300],
    [4, 400],
    [5, 500]
])

print("Original data:")
print(X)

# PCA without standardization
pca_no_scale = PCA(n_components=2)
X_no_scale = pca_no_scale.fit_transform(X)

print("\n--- Without Standardization ---")
print("Explained variance ratio:", pca_no_scale.explained_variance_ratio_)
print("PC1 explains: {:.2%}".format(pca_no_scale.explained_variance_ratio_[0]))

# PCA with standardization
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)
pca_scaled = PCA(n_components=2)
X_with_scale = pca_scaled.fit_transform(X_scaled)

print("\n--- With Standardization ---")
print("Explained variance ratio:", pca_scaled.explained_variance_ratio_)
print("PC1 explains: {:.2%}".format(pca_scaled.explained_variance_ratio_[0]))

print("\n--- Comparison ---")
print("Without scaling: PC1 dominates because feature 2 has much larger scale")
print("With scaling: Both features contribute more evenly to PCs")

# Visualization
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.scatter(X_no_scale[:, 0], X_no_scale[:, 1], color='blue', s=50)
plt.xlabel('PC1')
plt.ylabel('PC2')
plt.title('PCA Without Standardization')
plt.grid(True)

plt.subplot(1, 2, 2)
plt.scatter(X_with_scale[:, 0], X_with_scale[:, 1], color='red', s=50)
plt.xlabel('PC1')
plt.ylabel('PC2')
plt.title('PCA With Standardization')
plt.grid(True)

plt.tight_layout()
plt.show()
